        function escapeHtml(str) {
            if (typeof str !== "string") return "";
            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }

        function formatDateTime(ts) {
            if (!ts) return "-";
            const d = new Date(ts);
            const y = d.getFullYear();
            const m = String(d.getMonth() + 1).padStart(2, "0");
            const day = String(d.getDate()).padStart(2, "0");
            const hh = String(d.getHours()).padStart(2, "0");
            const mm = String(d.getMinutes()).padStart(2, "0");
            return `${y}.${m}.${day} ${hh}:${mm}`;
        }

        function speakChinese(text) {
            if (!("speechSynthesis" in window)) {
                showToast("이 브라우저에서는 발음 듣기를 지원하지 않습니다.");
                return;
            }
            try {
                window.speechSynthesis.cancel();
                const u = new SpeechSynthesisUtterance(text);
                u.lang = "zh-CN";
                u.rate = 0.92;
                const voices = window.speechSynthesis.getVoices();
                const zhVoice = voices.find(v => v.lang === "zh-CN" || v.lang === "zh_CN") || voices.find(v => v.lang.startsWith("zh"));
                if (zhVoice) u.voice = zhVoice;
                window.speechSynthesis.speak(u);
            } catch (e) {}
        }

        /**
         * 원본 사진 비율·내용을 100% 유지하면서 Firestore 1MB 제한 내로 안전하게 압축
         */
        function compressOriginalImageToDataUrl(file, maxDimension = 1280, maxBase64Length = 800000) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onerror = () => reject(new Error("파일을 읽을 수 없습니다."));
                reader.onload = (e) => {
                    const img = new Image();
                    img.onerror = () => reject(new Error("올바른 이미지 파일이 아닙니다."));
                    img.onload = () => {
                        let { width, height } = img;
                        if (width > maxDimension || height > maxDimension) {
                            const ratio = Math.min(maxDimension / width, maxDimension / height);
                            width = Math.round(width * ratio);
                            height = Math.round(height * ratio);
                        }

                        const canvas = document.createElement("canvas");
                        canvas.width = width;
                        canvas.height = height;
                        const ctx = canvas.getContext("2d");
                        ctx.drawImage(img, 0, 0, width, height);

                        let quality = 0.88;
                        let dataUrl = canvas.toDataURL("image/jpeg", quality);

                        // Firestore 문서 용량 제한(1MB) 안전선 확보
                        while (dataUrl.length > maxBase64Length && quality > 0.4) {
                            quality -= 0.08;
                            dataUrl = canvas.toDataURL("image/jpeg", quality);
                        }

                        // 여전히 크면 해상도를 한 단계 더 줄여 원본 비율 유지
                        if (dataUrl.length > maxBase64Length) {
                            const scale = 0.72;
                            canvas.width = Math.round(width * scale);
                            canvas.height = Math.round(height * scale);
                            ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
                            dataUrl = canvas.toDataURL("image/jpeg", 0.72);
                        }

                        resolve(dataUrl);
                    };
                    img.src = e.target.result;
                };
                reader.readAsDataURL(file);
            });
        }

        function initSnowCanvas() {
            const canvas = document.getElementById("snowCanvas");
            const ctx = canvas.getContext("2d");
            let flakes = [];

            function resize() {
                canvas.width = window.innerWidth;
                canvas.height = window.innerHeight;
                const count = Math.min(70, Math.floor((canvas.width * canvas.height) / 18000));
                flakes = Array.from({ length: count }, () => ({
                    x: Math.random() * canvas.width,
                    y: Math.random() * canvas.height,
                    r: Math.random() * 2.1 + 0.7,
                    speedY: Math.random() * 0.75 + 0.28,
                    speedX: Math.random() * 0.45 - 0.22,
                    opacity: Math.random() * 0.5 + 0.2
                }));
            }

            function draw() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                for (const f of flakes) {
                    ctx.beginPath();
                    ctx.arc(f.x, f.y, f.r, 0, Math.PI * 2);
                    ctx.fillStyle = `rgba(255, 250, 245, ${f.opacity})`;
                    ctx.fill();
                    f.y += f.speedY;
                    f.x += f.speedX + Math.sin(f.y * 0.01) * 0.18;
                    if (f.y > canvas.height + 5) {
                        f.y = -5;
                        f.x = Math.random() * canvas.width;
                    }
                    if (f.x > canvas.width + 5) f.x = -5;
                    if (f.x < -5) f.x = canvas.width + 5;
                }
                requestAnimationFrame(draw);
            }

            window.addEventListener("resize", resize);
            resize();
            draw();
        }

        // =========================================================================
        async function callGeminiJSON(payload, maxRetries = 3) {
            let delay = 1000;
            for (let attempt = 0; attempt <= maxRetries; attempt++) {
                try {
                    const res = await fetch(GEMINI_URL, {
                        method: "POST",
                        headers: { "Content-Type": "application/json" },
                        body: JSON.stringify(payload)
                    });
                    if (!res.ok) {
                        if ((res.status === 429 || res.status >= 500) && attempt < maxRetries) {
                            await new Promise(r => setTimeout(r, delay));
                            delay *= 2;
                            continue;
                        }
                        throw new Error(`HTTP ${res.status}`);
                    }
                    const json = await res.json();
                    const text = json?.candidates?.[0]?.content?.parts?.[0]?.text;
                    if (!text) throw new Error("Empty response");
                    const cleaned = text.replace(/^```json\s*/i, "").replace(/```\s*$/i, "").trim();
                    return JSON.parse(cleaned);
                } catch (err) {
                    if (attempt < maxRetries) {
                        await new Promise(r => setTimeout(r, delay));
                        delay *= 2;
                        continue;
                    }
                    throw err;
                }
            }
        }

        async function fetchFreshWordsFromGemini(level, easyNeed, medNeed, hardNeed, excludeWords) {
            const total = easyNeed + medNeed + hardNeed;
            if (total <= 0) return { easy: [], medium: [], hard: [] };

            const avoidList = excludeWords.slice(-60).join(", ");
            const payload = {
                systemInstruction: {
                    parts: [{
                        text: `당신은 HSK 1~9급 공식 어휘 전문가입니다. 반드시 중국어 간체자(简体字)만 사용하세요. 요청된 HSK ${level}급에 맞는 단어를 난이도별(easy ${easyNeed}개, medium ${medNeed}개, hard ${hardNeed}개)로 생성하세요. 각 예문(exCn)에는 반드시 해당 단어(word)가 그대로 포함되어야 합니다.`
                    }]
                },
                contents: [{
                    role: "user",
                    parts: [{
                        text: `HSK ${level}급 새 단어 총 ${total}개(easy ${easyNeed}개, medium ${medNeed}개, hard ${hardNeed}개)를 JSON으로 만들어 주세요. 제외 단어: [${avoidList}]`
                    }]
                }],
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: {
                        type: "OBJECT",
                        properties: {
                            items: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        diff: { type: "STRING", enum: ["easy", "medium", "hard"] },
                                        word: { type: "STRING" },
                                        pinyin: { type: "STRING" },
                                        meaning: { type: "STRING" },
                                        synonyms: { type: "ARRAY", items: { type: "STRING" } },
                                        exCn: { type: "STRING" },
                                        exKo: { type: "STRING" }
                                    },
                                    required: ["diff", "word", "pinyin", "meaning", "synonyms", "exCn", "exKo"]
                                }
                            }
                        },
                        required: ["items"]
                    }
                }
            };

            const res = await callGeminiJSON(payload, 2);
            const grouped = { easy: [], medium: [], hard: [] };
            if (Array.isArray(res?.items)) {
                for (const it of res.items) {
                    if (grouped[it.diff] && it.word && it.pinyin && it.meaning && it.exCn && it.exCn.includes(it.word)) {
                        grouped[it.diff].push([it.word, it.pinyin, it.meaning, it.synonyms || [], it.exCn, it.exKo || ""]);
                    }
                }
            }
            return grouped;
        }

        async function gradeMeaningWithGemini(word, pinyin, officialMeaning, synonyms, userAnswer) {
            const payload = {

        function shuffleArray(arr) {
            const copy = [...arr];
            for (let i = copy.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [copy[i], copy[j]] = [copy[j], copy[i]];
            }
            return copy;
        }

        function getDifficultyDistribution(count) {
            if (count === 5) return { easy: 2, medium: 2, hard: 1 };
            if (count === 20) return { easy: 7, medium: 7, hard: 6 };
            return { easy: 4, medium: 3, hard: 3 };
        }

        async function startQuizSession() {
            if (!currentUser) return;
            quizSetupPanel.classList.add("hidden");
            quizResultPanel.classList.add("hidden");
            quizLoadingPanel.classList.remove("hidden");

            const dist = getDifficultyDistribution(selectedCount);
            const bank = HSK_BANK[selectedLevel];
            const recentSet = new Set(currentUser.recentWords || []);

            const pickFromPool = (pool, need) => {
                const unseen = shuffleArray(pool.filter(w => !recentSet.has(w[0])));
                const seen = shuffleArray(pool.filter(w => recentSet.has(w[0])));
                return [...unseen, ...seen].slice(0, need);
            };

            let easyWords = pickFromPool(bank.easy, dist.easy);
            let medWords = pickFromPool(bank.medium, dist.medium);
            let hardWords = pickFromPool(bank.hard, dist.hard);

            if (aiExpandToggle.checked) {
                try {
                    const exclude = [
                        ...Array.from(recentSet),
                        ...easyWords.map(w => w[0]),
                        ...medWords.map(w => w[0]),
                        ...hardWords.map(w => w[0])
                    ];
                    const fresh = await fetchFreshWordsFromGemini(
                        selectedLevel,
                        Math.ceil(dist.easy / 2),
                        Math.ceil(dist.medium / 2),
                        Math.ceil(dist.hard / 2),
                        exclude
                    );
                    const mergeFresh = (base, freshArr, targetCount) => {
                        const used = new Set();
                        const merged = [];
                        for (const item of [...freshArr, ...base]) {
                            if (!used.has(item[0]) && merged.length < targetCount) {
                                used.add(item[0]);
                                merged.push(item);
                            }
                        }
                        return merged;
                    };
                    easyWords = mergeFresh(easyWords, fresh.easy, dist.easy);
                    medWords = mergeFresh(medWords, fresh.medium, dist.medium);
                    hardWords = mergeFresh(hardWords, fresh.hard, dist.hard);
                } catch (e) {}
            }

            const orderedRaw = [
                ...easyWords.map(w => ({ raw: w, diff: "easy" })),
                ...medWords.map(w => ({ raw: w, diff: "medium" })),
                ...hardWords.map(w => ({ raw: w, diff: "hard" }))
            ];

            const typeOffset = Math.floor(Math.random() * 4);
            activeQuestions = orderedRaw.map((entry, idx) => {
                const [word, pinyin, meaning, synonyms, exCn, exKo] = entry.raw;
                return {
                    diff: entry.diff,
                    points: DIFF_POINTS[entry.diff],
                    type: ((idx + typeOffset) % 4) + 1,
                    word,
                    pinyin,
                    meaning,
                    synonyms: Array.isArray(synonyms) ? synonyms : [],
                    exCn,
                    exKo
                };
            });

            currentUser.recentWords = Array.from(new Set([...(currentUser.recentWords || []), ...activeQuestions.map(q => q.word)])).slice(-120);
            syncCurrentUserToFirestore({ recentWords: currentUser.recentWords });

            currentQIndex = 0;
            sessionCorrect = 0;
            sessionPoints = 0;
            sessionDiffStats = {
                easy: { c: 0, t: dist.easy },
                medium: { c: 0, t: dist.medium },
                hard: { c: 0, t: dist.hard }
            };

            quizLoadingPanel.classList.add("hidden");
            quizActivePanel.classList.remove("hidden");
            renderCurrentQuestion();
        }

        function normalizeKo(s) {
            return s.trim().toLowerCase()
                .replace(/[~.,!?()\[\]\s'"`·]/g, "")
                .replace(/(하다|이다|되다|해요|어요|아요|다|요|을|를|이|가|은|는|의)$/g, "");
        }

        function normalizeZh(s) {
            return s.trim().replace(/[\s.,!?。，！？]/g, "");
        }

        async function handleAnswerSubmit(isSkip = false) {
            if (isGrading || isWaitingNext) return;
            const q = activeQuestions[currentQIndex];
            const userAns = qAnswerInput.value.trim();

            if (!isSkip && !userAns) {
                showToast("정답을 입력하거나 '모르겠어요'를 눌러주세요.");
                qAnswerInput.focus();
                return;
            }

            isGrading = true;
            qAnswerInput.disabled = true;
            qSubmitBtn.disabled = true;

            let isCorrect = false;
            let feedbackText = "";

            if (isSkip) {
                isCorrect = false;
                feedbackText = "괜찮아요! 린젠칭도 처음엔 다 서툴렀는걸요. 이번 기회에 꼭 기억해 두세요.";
            } else if (q.type === 2 || q.type === 4) {
                if (normalizeZh(userAns) === normalizeZh(q.word)) {
                    isCorrect = true;
                    feedbackText = "정확한 간체자 표기예요! 이언이 켈리에게 한 걸음 더 가까워졌어요.";
                } else {
                    isCorrect = false;
                    feedbackText = `아쉬워요! 입력하신 "${userAns}" 대신 간체자 "${q.word}"가 정답이에요.`;
                }
            } else {
                const candidates = [q.meaning, ...q.synonyms];
                const normUser = normalizeKo(userAns);
                const localMatch = candidates.some(c => {
                    const normC = normalizeKo(c);
                    return normUser === normC || (normUser.length >= 2 && normC.length >= 2 && (normC.includes(normUser) || normUser.includes(normC)));
                });

                if (localMatch) {
                    isCorrect = true;
                    feedbackText = "정확해요! 핵심 의미를 아주 깔끔하게 짚어내셨습니다.";
                } else {
                    qSubmitBtn.textContent = "AI 채점 중...";
                    try {
                        const aiGrade = await gradeMeaningWithGemini(q.word, q.pinyin, q.meaning, q.synonyms, userAns);
                        isCorrect = !!aiGrade.isCorrect;
                        feedbackText = aiGrade.reason || (isCorrect ? "유사한 의미로 인정해 드렸어요!" : "아쉽게도 뜻이 조금 달랐어요.");
                    } catch (e) {
                        isCorrect = false;
                        feedbackText = `대표 정답은 "${q.meaning}"입니다.`;
                    }
                }
            }

            if (isCorrect) {
                sessionCorrect++;
                sessionPoints += q.points;
                sessionDiffStats[q.diff].c++;
                await addPointsAndCheckTitles(q.points);
                triggerWarmBloom(2200);
            }

            qInputArea.classList.add("hidden");
            qFeedbackArea.classList.remove("hidden");

            if (isCorrect) {
                qFeedbackArea.className = "rounded-2xl p-5 border border-amber-400/50 bg-amber-500/10 space-y-3 animate-fade-up";
                qFeedbackIcon.textContent = "✨";
                qFeedbackTitle.textContent = "정답입니다! 세상에 색채가 번집니다";
                qFeedbackPoints.textContent = `+${q.points} P`;
                qFeedbackPoints.className = "px-3 py-1 rounded-full text-xs font-bold bg-amber-500/25 text-amber-200 border border-amber-400/40";
            } else {
                qFeedbackArea.className = "rounded-2xl p-5 border border-zinc-600/60 bg-zinc-900/80 space-y-3 animate-fade-up";
                qFeedbackIcon.textContent = "🎞️";
                qFeedbackTitle.textContent = isSkip ? "정답을 확인해 보세요" : "조금 아쉬워요 (오답)";
                qFeedbackPoints.textContent = "+0 P";
                qFeedbackPoints.className = "px-3 py-1 rounded-full text-xs font-bold bg-white/10 text-zinc-400 border border-white/10";
            }

            qFeedbackMsg.textContent = feedbackText;
            qReviewWord.textContent = q.word;
            qReviewPinyin.textContent = q.pinyin;
            qReviewMeaning.textContent = q.meaning + (q.synonyms.length ? ` (${q.synonyms.slice(0, 3).join(", ")})` : "");
            qReviewExample.textContent = q.exCn;
            qReviewExampleKo.textContent = q.exKo;

            isGrading = false;
            isWaitingNext = true;
            qNextBtn.focus();
        }

                const file = files[i];

                    const dataUrl = await compressOriginalImageToDataUrl(file, 1280, 780000);

                        title: cleanName || `추억 사진`,
                        comment: "",
                        createdAt: Date.now() + i
